function send(channel: string, ...args: unknown[]) {
  if (mainWindow && !mainWindow.isDestroyed()) {

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1500,
    height: 950,
    minWidth: 1100,
    minHeight: 700,
    backgroundColor: '#0d1117',
    show: false,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: false,
      spellcheck: false,
    },
  });

  mainWindow.once('ready-to-show', () => mainWindow?.show());

  if (isDev) {
    mainWindow.loadURL('http://localhost:5173');
    mainWindow.webContents.openDevTools({ mode: 'detach' });
  } else {
    mainWindow.loadFile(path.join(__dirname, '../dist/index.html'));
  }

  mainWindow.webContents.setWindowOpenHandler(({ url }) => {
    if (/^https?:/.test(url)) shell.openExternal(url);
    return { action: 'deny' };
  });

  mainWindow.webContents.on('will-navigate', (e, url) => {
    const ok = url.startsWith('http://localhost:5173') || url.startsWith('file://');
    if (!ok) {
      e.preventDefault();
      shell.openExternal(url);
    }
  });

  mainWindow.on('closed', () => (mainWindow = null));
}

type Unsub = () => void;

const MENU_CHANNELS = [
  'openFolder', 'save', 'palette', 'sidebar', 'chat', 'terminal', 'ask',
] as const;

const api = {
  app: {
    info: () => ipcRenderer.invoke('app:info'),
    onMenu: (cb: (action: string) => void): Unsub => {
      const subs: Array<[string, (...a: any[]) => void]> = [];
      for (const c of MENU_CHANNELS) {
        const ch = `menu:${c}`;
        const h = () => cb(c);
        ipcRenderer.on(ch, h);
        subs.push([ch, h]);
      }
      return () => {
        for (const [ch, h] of subs) ipcRenderer.removeListener(ch, h);
      };
    },
  },

  fs: {
    readDir: (p: string) => ipcRenderer.invoke('fs:readDir', p),
    readFile: (p: string) => ipcRenderer.invoke('fs:readFile', p),
    writeFile: (p: string, c: string) => ipcRenderer.invoke('fs:writeFile', p, c),
    createFile: (p: string) => ipcRenderer.invoke('fs:createFile', p),
    createDir: (p: string) => ipcRenderer.invoke('fs:createDir', p),
    delete: (p: string) => ipcRenderer.invoke('fs:delete', p),
    rename: (o: string, n: string) => ipcRenderer.invoke('fs:rename', o, n),
    watch: (p: string, cb: (evt: string, file: string) => void): Unsub => {
      const channel = `fs:watch:${p}`;
      const handler = (_e: unknown, evt: string, file: string) => cb(evt, file);
      ipcRenderer.on(channel, handler);
      ipcRenderer.send('fs:watch:start', p);
      return () => {
        ipcRenderer.removeListener(channel, handler);
        ipcRenderer.send('fs:watch:stop', p);
      };
    },
  },

  dialog: {
    openDirectory: () => ipcRenderer.invoke('dialog:openDirectory'),
  },

  ollama: {
    health: () => ipcRenderer.invoke('ollama:health'),
    list: () => ipcRenderer.invoke('ollama:list'),
    chat: (payload: unknown) => ipcRenderer.invoke('ollama:chat', payload),
    stream: (
      payload: { model: string; messages: unknown[]; temperature?: number },
      onChunk: (c: string) => void,
      onDone: () => void,
      onError?: (e: string) => void
    ): Unsub => {
      const id = Math.random().toString(36).slice(2) + Date.now().toString(36);
      const chunkCh = `ollama:chunk:${id}`;
      const doneCh = `ollama:done:${id}`;
      const errCh = `ollama:error:${id}`;

      let finished = false;

      const cleanup = () => {
        ipcRenderer.removeListener(chunkCh, onChunkH);
        ipcRenderer.removeListener(doneCh, onDoneH);
        ipcRenderer.removeListener(errCh, onErrH);
      };
      const onChunkH = (_e: unknown, c: string) => onChunk(c);
      const onDoneH = () => { if (!finished) { finished = true; onDone(); cleanup(); } };
      const onErrH = (_e: unknown, msg: string) => {
        if (!finished) { finished = true; onError?.(msg); cleanup(); }
      };

      ipcRenderer.on(chunkCh, onChunkH);
      ipcRenderer.on(doneCh, onDoneH);
      ipcRenderer.on(errCh, onErrH);

      ipcRenderer.send('ollama:stream', { ...payload, id });

      return () => {
        finished = true;
        cleanup();
        ipcRenderer.send('ollama:abort', id);
      };
    },
  },

  terminal: {
    create: (id: string, cwd: string) => ipcRenderer.invoke('term:create', id, cwd),
    write: (id: string, data: string) => ipcRenderer.send('term:write', id, data),
    resize: (id: string, cols: number, rows: number) =>
      ipcRenderer.send('term:resize', id, cols, rows),
    onData: (id: string, cb: (d: string) => void): Unsub => {

    },
    kill: (id: string) => ipcRenderer.send('term:kill', id),

  project: {

  },

import type { IpcMain } from 'electron';
import fs from 'node:fs/promises';
import path from 'node:path';
import chokidar, { type FSWatcher } from 'chokidar';

const watchers = new Map<string, FSWatcher>();

const IGNORE = new Set([
  'node_modules', '.git', 'dist', 'build', '.next', '__pycache__',
  '.venv', 'venv', '.idea', '.vscode', 'target', 'out', '.cache',
  '.turbo', 'coverage', '.pytest_cache', '.mypy_cache',
]);

const MAX_READ = 10 * 1024 * 1024;

function isIgnored(p: string) {
  return p.split(/[\\/]/).some((seg) => IGNORE.has(seg));
}

export function registerFsHandlers(ipc: IpcMain) {
  ipc.handle('fs:readDir', async (_e, dir: string) => {
    const entries = await fs.readdir(dir, { withFileTypes: true });
    const result = await Promise.all(
      entries
        .filter((e) => !IGNORE.has(e.name))
        .map(async (e) => {
          const full = path.join(dir, e.name);
          let size = 0, mtime = 0;
          try {
            const st = await fs.stat(full);
            size = st.size;
            mtime = st.mtimeMs;
          } catch {}
          return { name: e.name, path: full, isDirectory: e.isDirectory(), size, mtime };
        })
    );
    return result.sort((a, b) =>
      a.isDirectory === b.isDirectory
        ? a.name.localeCompare(b.name)
        : a.isDirectory ? -1 : 1
    );
  });

  ipc.handle('fs:readFile', async (_e, p: string) => {
    const st = await fs.stat(p);
    if (st.size > MAX_READ) throw new Error(`File too large (>${MAX_READ / 1024 / 1024}MB)`);
    return fs.readFile(p, 'utf-8');
  });

  ipc.handle('fs:writeFile', async (_e, p: string, content: string) => {
    const dir = path.dirname(p);
    await fs.mkdir(dir, { recursive: true });
    const tmp = `${p}.tmp-${process.pid}-${Date.now()}`;
    await fs.writeFile(tmp, content, 'utf-8');
    await fs.rename(tmp, p);
    return true;
  });

  ipc.handle('fs:createFile', async (_e, p: string) => {
    await fs.writeFile(p, '', { flag: 'wx' });
    return true;
  });

  ipc.handle('fs:createDir', async (_e, p: string) => {
    await fs.mkdir(p, { recursive: true });
    return true;
  });

  ipc.handle('fs:delete', async (_e, p: string) => {
    await fs.rm(p, { recursive: true, force: true });
    return true;
  });

  ipc.handle('fs:rename', async (_e, oldP: string, newP: string) => {
    await fs.rename(oldP, newP);
    return true;
  });

  ipc.on('fs:watch:start', (e, dir: string) => {
    if (watchers.has(dir)) return;
    const w = chokidar.watch(dir, {
      ignored: (p) => isIgnored(p),
      ignoreInitial: true,
      depth: 10,
      awaitWriteFinish: { stabilityThreshold: 150 },
    });
    w.on('all', (evt, file) => {
      if (!e.sender.isDestroyed()) e.sender.send(`fs:watch:${dir}`, evt, file);
    });
    watchers.set(dir, w);
  });

  ipc.on('fs:watch:stop', (_e, dir: string) => {
    watchers.get(dir)?.close();
    watchers.delete(dir);
  });
}

import type { IpcMain } from 'electron';
import http from 'node:http';

const OLLAMA = 'http://127.0.0.1:11434';
const activeRequests = new Map<string, http.ClientRequest>();

function request(url: string, method: string, body?: unknown): Promise<any> {
  return new Promise((resolve, reject) => {
    const u = new URL(url);
    const req = http.request(
      {
        hostname: u.hostname,
        port: u.port,
        path: u.pathname + u.search,
        method,
        timeout: 5000,
        headers: body ? { 'Content-Type': 'application/json' } : {},
      },
      (res) => {
        let data = '';
        res.on('data', (c) => (data += c));
        res.on('end', () => {
          try { resolve(JSON.parse(data)); } catch { resolve(data); }
        });
      }
    );
    req.on('timeout', () => req.destroy(new Error('Ollama request timed out')));
    req.on('error', reject);
    if (body) req.write(JSON.stringify(body));
    req.end();
  });
}

export function registerOllamaHandlers(ipc: IpcMain) {
  ipc.handle('ollama:health', async () => {
    try {
      await request(`${OLLAMA}/api/tags`, 'GET');
      return { ok: true, url: OLLAMA };
    } catch (e) {
      return { ok: false, url: OLLAMA, error: (e as Error).message };
    }
  });

  ipc.handle('ollama:list', async () => {
    try {
      const res = await request(`${OLLAMA}/api/tags`, 'GET');
      return res.models ?? [];
    } catch {
      return [];
    }
  });

  ipc.handle('ollama:chat', async (_e, payload: any) => {
    return request(`${OLLAMA}/api/chat`, 'POST', {
      model: payload.model,
      messages: payload.messages,
      stream: false,
      options: { temperature: payload.temperature ?? 0.7 },
    });
  });

  ipc.on('ollama:stream', (e, payload: any) => {
    const u = new URL(`${OLLAMA}/api/chat`);
    let finished = false;
    const finalize = () => {
      if (!finished) {
        finished = true;
        if (!e.sender.isDestroyed()) e.sender.send(`ollama:done:${payload.id}`);
      }
      activeRequests.delete(payload.id);
    };

    const req = http.request(
      {
        hostname: u.hostname,
        port: u.port,
        path: u.pathname,
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
      },
      (res) => {
        let buf = '';
        res.on('data', (chunk: Buffer) => {
          buf += chunk.toString('utf-8');
          const parts = buf.split('\n');
          buf = parts.pop() ?? '';
          for (const line of parts) {
            if (!line.trim()) continue;
            try {
              const obj = JSON.parse(line);
              if (obj.message?.content && !e.sender.isDestroyed()) {
                e.sender.send(`ollama:chunk:${payload.id}`, obj.message.content);
              }
              if (obj.done) finalize();
            } catch { /* partial */ }
          }
        });
        res.on('end', finalize);
      }
    );

    req.on('error', (err) => {
      if (!e.sender.isDestroyed()) e.sender.send(`ollama:error:${payload.id}`, err.message);
      finished = true;
      activeRequests.delete(payload.id);
    });

    req.write(
      JSON.stringify({
        model: payload.model,
        messages: payload.messages,
        stream: true,
        options: { temperature: payload.temperature ?? 0.7 },
      })
    );
    req.end();
    activeRequests.set(payload.id, req);
  });

  ipc.on('ollama:abort', (_e, id: string) => {
    const r = activeRequests.get(id);
    if (r) {
      r.destroy();
      activeRequests.delete(id);
    }
  });
}

import type { IpcMain } from 'electron';
import * as pty from 'node-pty';
import os from 'node:os';
import fs from 'node:fs';

const terminals = new Map<string, pty.IPty>();

function pickShell(): string {
  if (process.platform === 'win32') {
    const candidates = [
      'C:\\Program Files\\PowerShell\\7\\pwsh.exe',
      'C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe',
      process.env.COMSPEC,
    ].filter(Boolean) as string[];
    for (const c of candidates) {
      try { if (fs.existsSync(c)) return c; } catch {}
    }
    return process.env.COMSPEC || 'cmd.exe';
  }
  return process.env.SHELL || '/bin/bash';
}

export function registerTerminalHandlers(ipc: IpcMain) {
  ipc.handle('term:create', (e, id: string, cwd: string) => {
    if (terminals.has(id)) return true;
    let safeCwd = cwd || os.homedir();
    try { if (!fs.existsSync(safeCwd)) safeCwd = os.homedir(); } catch { safeCwd = os.homedir(); }

    const shell = pickShell();
    const args = process.platform === 'win32' && shell.endsWith('powershell.exe')
      ? ['-NoLogo']
      : [];

    const p = pty.spawn(shell, args, {
      name: 'xterm-256color',
      cols: 80,
      rows: 24,
      cwd: safeCwd,
      env: { ...process.env, TERM: 'xterm-256color' } as Record<string, string>,
    });

    p.onData((d) => {
      if (!e.sender.isDestroyed()) e.sender.send(`term:data:${id}`, d);
    });
    p.onExit(({ exitCode }) => {
      terminals.delete(id);
      if (!e.sender.isDestroyed()) {
        e.sender.send(`term:data:${id}`, `\r\n[process exited with code ${exitCode}]\r\n`);
      }
    });

    terminals.set(id, p);
    return true;
  });

  ipc.on('term:write', (_e, id: string, data: string) => {
    terminals.get(id)?.write(data);
  });

  ipc.on('term:resize', (_e, id: string, cols: number, rows: number) => {
    try { terminals.get(id)?.resize(Math.max(2, cols), Math.max(2, rows)); } catch {}
  });

  ipc.on('term:kill', (_e, id: string) => {
    try { terminals.get(id)?.kill(); } catch {}
    terminals.delete(id);
  });
}

import type { IpcMain } from 'electron';
import fs from 'node:fs/promises';
import path from 'node:path';

const IGNORE = new Set([
  'node_modules', '.git', 'dist', 'build', '.next', '__pycache__',
  '.venv', 'venv', '.idea', '.vscode', 'target', 'out', '.cache', 'coverage',
]);

const TEXT_EXT = new Set([
  '.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs', '.json', '.md', '.txt',
  '.html', '.css', '.scss', '.less', '.py', '.rb', '.go', '.rs', '.java',
  '.kt', '.swift', '.c', '.h', '.cpp', '.hpp', '.cs', '.php', '.sh',
  '.yml', '.yaml', '.toml', '.ini', '.sql', '.vue', '.svelte', '.env',
]);

const MAX_FILE = 1 * 1024 * 1024;
const MAX_RESULTS = 500;

interface Hit { file: string; line: number; text: string; }

async function walk(dir: string, out: string[], budget = { count: 0 }) {
  if (budget.count > 5000) return;
  let entries: import('node:fs').Dirent[];
  try {
    entries = await fs.readdir(dir, { withFileTypes: true });
  } catch { return; }

  for (const ent of entries) {
    if (IGNORE.has(ent.name)) continue;
    const full = path.join(dir, ent.name);
    if (ent.isDirectory()) {
      await walk(full, out, budget);
    } else if (ent.isFile()) {
      const ext = path.extname(ent.name).toLowerCase();
      if (TEXT_EXT.has(ext)) {
        out.push(full);
        budget.count++;
      }
    }
  }
}

export function registerSearchHandlers(ipc: IpcMain) {
  ipc.handle('search:inWorkspace', async (_e, root: string, query: string, opts?: { max?: number }) => {
    if (!query || !root) return [];
    const max = opts?.max ?? MAX_RESULTS;
    const files: string[] = [];
    await walk(root, files);
    const q = query.toLowerCase();
    const hits: Hit[] = [];

    for (const file of files) {
      if (hits.length >= max) break;
      try {
        const st = await fs.stat(file);
        if (st.size > MAX_FILE) continue;
        const text = await fs.readFile(file, 'utf-8');
        const lines = text.split(/\r?\n/);
        for (let i = 0; i < lines.length; i++) {
          if (lines[i].toLowerCase().includes(q)) {
            hits.push({ file, line: i + 1, text: lines[i].slice(0, 300) });
            if (hits.length >= max) break;
          }
        }
      } catch {}
    }
    return hits;
  });
}

function gitFor(root: string) {

export function registerGitHandlers(ipc: IpcMain) {
  ipc.handle('git:status', async (_e, root: string) => {
    try {
      const g = gitFor(root);
      const isRepo = await g.checkIsRepo();
      if (!isRepo) return { isRepo: false, files: [] };
      const status = await g.status();
      const files = status.files.map((f) => ({
        path: f.path,
        index: f.index,
        working_dir: f.working_dir,
      }));
      return { isRepo: true, branch: status.current, files };
    } catch (e) {
      return { isRepo: false, files: [], error: (e as Error).message };
    }
  });

  ipc.handle('git:diff', async (_e, root: string, file: string) => {
    try {
      const g = gitFor(root);
      return await g.diff([file]);
    } catch (e) {
      return '';
    }
  });
}

    setFiles(dir, list);
    return list;
  }, [setFiles]);

    const dir = await (window as any).applab.dialog.openDirectory();
    if (!dir) return;
    setWorkspace(dir);
    await loadDir(dir);
    setExpanded({ [dir]: true });
    refreshGit(dir);

    const unwatch = (window as any).applab.fs.watch(dir, async (evt: string, file: string) => {
      if (['add', 'unlink', 'addDir', 'unlinkDir'].includes(evt)) {
        await loadDir(dir);
        refreshGit(dir);
      } else if (evt === 'change') {
        refreshGit(dir);
        // Reload tab if file is not dirty
        const state = useStore.getState();
        const tab = state.openTabs.find((t) => t.path === file);
        if (tab && !tab.dirty) {
          try {
            const content = await (window as any).applab.fs.readFile(file);
            state.openFile(file, content);
          } catch {}
        }
      }
    });
    return unwatch;
  }, [loadDir, refreshGit, setWorkspace]);

  useEffect(() => {
    const handler = () => pickFolder();
    window.addEventListener('applab:openFolder', handler);
    return () => window.removeEventListener('applab:openFolder', handler);
  }, [pickFolder]);

  const handleClick = async (node: { path: string; isDirectory: boolean }) => {
    if (node.isDirectory) {
      const isOpen = expanded[node.path];
      setExpanded((e) => ({ ...e, [node.path]: !isOpen }));
      if (!isOpen && !files[node.path]) {
        await loadDir(node.path);
      }
    } else {
      try {
        const content = await (window as any).applab.fs.readFile(node.path);
        openFile(node.path, content);
      } catch (e) {
        alert(`Failed to open file: ${(e as Error).message}`);
      }
    }
  };

  const badgeFor = (path: string) => {
    const s = gitFiles[path];
    if (!s) return null;
    if (s.includes('?')) return 'U';
    if (s.includes('A')) return 'A';
    if (s.includes('M')) return 'M';
    if (s.includes('D')) return 'D';
    return '*';

  const renderTree = (dir: string, depth = 0): JSX.Element[] => {

            <span className="tree-name">{n.name}</span>
            {badge && <span className={`git-badge git-${badge}`}>{badge}</span>}
          </div>
          {n.isDirectory && isOpen && files[n.path] && renderTree(n.path, depth + 1)}
        </div>
      );
    });

      <div className="sidebar-header">

import { useEffect, useRef, useState } from 'react';
import { useStore } from '../../store';
import { buildSystemPrompt, QUICK_PROMPTS } from '../../ai/prompts';

function extractCodeBlock(text: string): string | null {
  const m = text.match(/```[a-zA-Z0-9+#-]*\n([\s\S]*?)```/);
  return m ? m[1] : null;
}

export default function ChatPanel() {
  const {
    activeModel, models, setActiveModel,
    messages, addMessage, appendToLast, setMessages,
    activeTab, openTabs, updateContent,
  } = useStore();
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);
  const abortRef = useRef<(() => void) | null>(null);
  const scrollRef = useRef<HTMLDivElement>(null);
  const workspace = useStore((s) => s.workspace);

  useEffect(() => {
    if (scrollRef.current) {
      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
    }
  }, [messages]);

  useEffect(() => {
    if (!workspace) return;
    (async () => {
      const saved = await (window as any).applab.project.loadChat(workspace);
      if (Array.isArray(saved) && saved.length) setMessages(saved);
    })();
  }, [workspace, setMessages]);

  const persist = (next: unknown[]) => {
    if (workspace) (window as any).applab.project.saveChat(workspace, next);
  };

  const send = async (overrideText?: string) => {
    const text = (overrideText ?? input).trim();
    if (!text || !activeModel || busy) return;

    const activeFile = openTabs.find((t) => t.path === activeTab);
    const userMsg = { role: 'user' as const, content: text };
    const history = [
      { role: 'system', content: buildSystemPrompt(activeFile?.path, activeFile?.content) },
      ...messages.map((m) => ({ role: m.role, content: m.content })),
      userMsg,
    ];
    const nextMessages = [...messages, userMsg, { role: 'assistant' as const, content: '' }];

    addMessage(userMsg);
    addMessage({ role: 'assistant', content: '' });
    persist(nextMessages);
    setInput('');
    setBusy(true);

    abortRef.current = (window as any).applab.ollama.stream(
      { model: activeModel, messages: history },
      (chunk: string) => appendToLast(chunk),
      () => {
        setBusy(false);
        abortRef.current = null;
        persist(useStore.getState().messages);
      },
      (err: string) => {
        appendToLast(`\n[error: ${err}]`);
        setBusy(false);
        abortRef.current = null;
      }
    );
  };

  const stop = () => {
    abortRef.current?.();
    abortRef.current = null;
    setBusy(false);
  };

  const applyCode = (msg: { content: string }) => {
    const code = extractCodeBlock(msg.content);
    if (!code) return alert('No fenced code block found in this message.');
    if (!activeTab) return alert('No file open. Open a file first.');
    const name = activeTab.split(/[\\/]/).pop();
    if (!confirm(`Replace contents of ${name} with the code block from this message?`)) return;
    updateContent(activeTab, code);
  };

  const quick = (key: keyof typeof QUICK_PROMPTS) => send(QUICK_PROMPTS[key]);

  return (
    <div className="chat-panel">
      <div className="chat-header">
        <select
          value={activeModel ?? ''}
          onChange={(e) => setActiveModel(e.target.value)}
        >
          {models.length === 0 && <option value="">No models found</option>}
          {models.map((m) => (
            <option key={m.name} value={m.name}>{m.name}</option>
          ))}
        </select>
        <div className="quick-row">
          <button onClick={() => quick('explain')} disabled={busy}>Explain</button>
          <button onClick={() => quick('refactor')} disabled={busy}>Refactor</button>
          <button onClick={() => quick('bugs')} disabled={busy}>Bugs</button>
          <button onClick={() => quick('tests')} disabled={busy}>Tests</button>
        </div>
      </div>

      <div className="messages" ref={scrollRef}>
        {messages.length === 0 && (
          <div className="chat-empty">
            Ask anything about your code. Open a file first for context-aware answers.
          </div>
        )}
        {messages.map((m, i) => (
          <div key={i} className={`msg msg-${m.role}`}>
            <div className="msg-role">{m.role}</div>
            <pre className="msg-content">{m.content || (busy ? '...' : '')}</pre>
            {m.role === 'assistant' && m.content.includes('```') && (
              <div className="msg-actions">
                <button onClick={() => applyCode(m)}>Apply to file</button>
                <button onClick={() => navigator.clipboard.writeText(m.content)}>Copy</button>
              </div>
            )}
          </div>
        ))}
      </div>

      <div className="chat-input">
        <textarea
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault();
              send();
            }
          }}
          placeholder="Ask AI about your code... (Enter to send, Shift+Enter for newline)"
        />
        <div className="chat-buttons">
          {busy
            ? <button onClick={stop} className="stop-btn">Stop</button>
            : <button onClick={() => send()} disabled={!activeModel}>Send</button>}
        </div>
      </div>
    </div>
  );
}

import { useCallback, useEffect, useRef, useState } from 'react';
import { Terminal } from '@xterm/xterm';
import { FitAddon } from '@xterm/addon-fit';
import '@xterm/xterm/css/xterm.css';
import { useStore } from '../../store';

interface TabState {
  id: string;
  term: Terminal;
  fit: FitAddon;
  dispose: () => void;
}

let tabCounter = 0;

export default function TerminalPanel() {
  const workspace = useStore((s) => s.workspace);
  const hostRef = useRef<HTMLDivElement>(null);
  const tabsRef = useRef<Map<string, TabState>>(new Map());
  const [tabs, setTabs] = useState<string[]>([]);
  const [active, setActive] = useState<string | null>(null);

  const attach = useCallback((id: string, tab: TabState) => {
    if (!hostRef.current) return;
    hostRef.current.innerHTML = '';
    hostRef.current.appendChild(tab.term.element!);
    requestAnimationFrame(() => {
      tab.fit.fit();
      (window as any).applab.terminal.resize(id, tab.term.cols, tab.term.rows);
    });
  }, []);

  const createTab = useCallback(async () => {
    if (!hostRef.current) return;
    const id = `t${++tabCounter}-${Math.random().toString(36).slice(2, 6)}`;
    const term = new Terminal({
      fontFamily: 'Consolas, "Cascadia Mono", monospace',
      fontSize: 13,
      theme: { background: '#0d1117', foreground: '#e6edf3' },
      cursorBlink: true,
    });
    const fit = new FitAddon();
    term.loadAddon(fit);
    term.open(hostRef.current);

    const offData = (window as any).applab.terminal.onData(id, (d: string) => term.write(d));
    term.onData((d) => (window as any).applab.terminal.write(id, d));
    term.onResize(({ cols, rows }) => (window as any).applab.terminal.resize(id, cols, rows));

    await (window as any).applab.terminal.create(id, workspace ?? '');

    const dispose = () => {
      offData();
      (window as any).applab.terminal.kill(id);
      term.dispose();
    };

    const tab: TabState = { id, term, fit, dispose };
    tabsRef.current.set(id, tab);
    setTabs((prev) => [...prev, id]);
    setActive(id);
    attach(id, tab);
  }, [workspace, attach]);

  const closeTab = (id: string) => {
    const tab = tabsRef.current.get(id);
    if (!tab) return;
    tab.dispose();
    tabsRef.current.delete(id);
    setTabs((prev) => {
      const next = prev.filter((x) => x !== id);
      if (active === id) {
        const newActive = next[next.length - 1] ?? null;
        setActive(newActive);
        if (newActive) {
          const t = tabsRef.current.get(newActive);
          if (t) attach(newActive, t);
        } else {
          hostRef.current && (hostRef.current.innerHTML = '');
        }
      }
      return next;
    });
  };

  const switchTo = (id: string) => {
    const tab = tabsRef.current.get(id);
    if (!tab) return;
    setActive(id);
    attach(id, tab);
  };

  useEffect(() => {
    createTab();
    const onResize = () => {
      const t = active ? tabsRef.current.get(active) : null;
      if (t) t.fit.fit();
    };
    window.addEventListener('resize', onResize);
    return () => {
      window.removeEventListener('resize', onResize);
      for (const t of tabsRef.current.values()) t.dispose();
      tabsRef.current.clear();
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  return (
    <div className="terminal-wrap">
      <div className="terminal-tabs">
        {tabs.map((id) => (
          <div
            key={id}
            className={`term-tab ${id === active ? 'active' : ''}`}
            onClick={() => switchTo(id)}
          >
            <span>shell</span>
            <button onClick={(e) => { e.stopPropagation(); closeTab(id); }}>×</button>
          </div>
        ))}
        <button className="term-add" onClick={createTab}>+</button>
      </div>
      <div ref={hostRef} className="terminal-xterm" />
    </div>
  );
}

import { useEffect, useMemo, useState } from 'react';
import { useStore } from '../../store';

interface Cmd {
  id: string;
  label: string;
  run: () => void;
}

export default function CommandPalette() {
  const {
    setPaletteOpen, toggleSidebar, toggleChat, toggleTerminal,
    openTabs, setActiveTab, closeTab, workspace,
  } = useStore();
  const [query, setQuery] = useState('');
  const [idx, setIdx] = useState(0);

  const commands: Cmd[] = useMemo(() => {
    const list: Cmd[] = [
      { id: 'open-folder', label: 'Open Folder...', run: () => window.dispatchEvent(new CustomEvent('applab:openFolder')) },
      { id: 'save', label: 'Save Current File', run: () => window.dispatchEvent(new CustomEvent('applab:save')) },
      { id: 'toggle-sidebar', label: 'Toggle Sidebar', run: toggleSidebar },
      { id: 'toggle-chat', label: 'Toggle AI Chat', run: toggleChat },
      { id: 'toggle-terminal', label: 'Toggle Terminal', run: toggleTerminal },
    ];
    for (const t of openTabs) {
      const name = t.path.split(/[\\/]/).pop() ?? t.path;
      list.push({ id: `open:${t.path}`, label: `Open: ${name}`, run: () => setActiveTab(t.path) });
      list.push({ id: `close:${t.path}`, label: `Close: ${name}`, run: () => closeTab(t.path) });
    }
    return list;
  }, [openTabs, toggleSidebar, toggleChat, toggleTerminal, setActiveTab, closeTab]);

  const filtered = useMemo(() => {
    if (!query.trim()) return commands;
    const q = query.toLowerCase();
    return commands.filter((c) => c.label.toLowerCase().includes(q));
  }, [commands, query]);

  useEffect(() => setIdx(0), [query]);

  const runSelected = () => {
    const cmd = filtered[idx];
    if (cmd) {
      cmd.run();
      setPaletteOpen(false);
    }
  };

  return (
    <div className="palette-overlay" onClick={() => setPaletteOpen(false)}>
      <div className="palette" onClick={(e) => e.stopPropagation()}>
        <input
          autoFocus
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Type a command..."
          onKeyDown={(e) => {
            if (e.key === 'ArrowDown') { e.preventDefault(); setIdx((i) => Math.min(i + 1, filtered.length - 1)); }
            if (e.key === 'ArrowUp') { e.preventDefault(); setIdx((i) => Math.max(i - 1, 0)); }
            if (e.key === 'Enter') { e.preventDefault(); runSelected(); }
          }}
        />
        <div className="palette-list">
          {filtered.length === 0 && <div className="palette-empty">No matching commands</div>}
          {filtered.map((c, i) => (
            <div
              key={c.id}
              className={`palette-item ${i === idx ? 'active' : ''}`}
              onMouseEnter={() => setIdx(i)}
              onClick={() => { c.run(); setPaletteOpen(false); }}
            >
              {c.label}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
